const SHEET_NAME = 'Fichas';

function doPost(e) {
  try {
    const data = JSON.parse(e.postData.contents);
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const sheet = ss.getSheetByName(SHEET_NAME);

    if (data.action === 'salvarFicha') {
      const f = data.ficha;
      const comps = (f.competencias||[]).map(c=>c.nome+':'+c.val).join('|');
      sheet.appendRow([
        f.id, f.nome, f.cargo, f.cbo, f.cboTitulo, f.depto,
        f.dataAdmissao, f.entrevistador, f.dataEntrevista,
        f.nivel, f.autonomia, f.formacao, f.experiencia, f.idiomas,
        f.atribuicoesCompete, f.atribuicoesAlem, f.atribuicoesNaoFaz,
        f.atividades, f.frequencia, f.cargaHoraria,
        f.responsabilidades, comps,
        f.pontosFortes, f.areasDesenvolvimento, f.expectativaCarreira,
        f.interesse, f.disponibilidade,
        f.data, f.status, '', '', ''
      ]);
    }

    if (data.action === 'atualizarFicha') {
      const sheet = ss.getSheetByName(SHEET_NAME);
      const rows = sheet.getDataRange().getValues();
      for (let i = 1; i < rows.length; i++) {
        if (String(rows[i][0]) === String(data.id)) {
          const campos = data.campos;
          const col = {
            status: 29, pdi: 30, prazoCtrl: 31, obsRH: 32
          };
          for (const [k, v] of Object.entries(campos)) {
            if (col[k]) sheet.getRange(i+1, col[k]).setValue(v);
          }
          break;
        }
      }
    }

    // Proxy seguro para a IA (Claude). A chave fica só aqui no servidor,
    // nunca no HTML/JS que roda no navegador do usuário.
    // OBS: a consulta à IA é feita via GET (doGet), não aqui — POST em
    // Apps Script tem problema conhecido de CORS no redirecionamento
    // da resposta, que impede o navegador de ler o retorno.

    return ContentService
      .createTextOutput(JSON.stringify({ok:true}))
      .setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService
      .createTextOutput(JSON.stringify({ok:false,error:err.message}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

// Chama a API da Anthropic usando a chave guardada em
// Configurações do projeto → Propriedades do script → ANTHROPIC_API_KEY
function chamarClaude(prompt, maxTokens) {
  const apiKey = PropertiesService.getScriptProperties().getProperty('ANTHROPIC_API_KEY');
  if (!apiKey) throw new Error('ANTHROPIC_API_KEY não configurada nas Propriedades do script.');
  const resp = UrlFetchApp.fetch('https://api.anthropic.com/v1/messages', {
    method: 'post',
    contentType: 'application/json',
    headers: { 'x-api-key': apiKey, 'anthropic-version': '2023-06-01' },
    payload: JSON.stringify({
      model: 'claude-sonnet-4-5-20250929',
      max_tokens: maxTokens,
      messages: [{ role: 'user', content: prompt }]
    }),
    muteHttpExceptions: true
  });
  const json = JSON.parse(resp.getContentText());
  if (json.error) throw new Error(json.error.message || 'Erro na API da Anthropic');
  return (json.content || []).map(function(b){ return b.text || ''; }).join('').trim();
}

function rowToFicha_(r) {
  const comps = String(r[21]||'').split('|').filter(Boolean).map(c=>{
    const [nome,val] = c.split(':');
    return {nome, val:parseInt(val)||0};
  });
  return {
    id:String(r[0]), nome:r[1], cargo:r[2], cbo:r[3], cboTitulo:r[4],
    depto:r[5], dataAdmissao:r[6], entrevistador:r[7], dataEntrevista:r[8],
    nivel:r[9], autonomia:r[10], formacao:r[11], experiencia:r[12], idiomas:r[13],
    atribuicoesCompete:r[14], atribuicoesAlem:r[15], atribuicoesNaoFaz:r[16],
    atividades:r[17], frequencia:r[18], cargaHoraria:r[19],
    responsabilidades:r[20], competencias:comps,
    pontosFortes:r[22], areasDesenvolvimento:r[23], expectativaCarreira:r[24],
    interesse:r[25], disponibilidade:r[26],
    data:r[27], status:r[28]||'novo', pdi:r[29]||'',
    prazoCtrl:r[30]||'', obsRH:r[31]||''
  };
}

const NIVEL_LABELS_ = ['Iniciante','Básico','Intermediário','Avançado','Expert'];

function construirPromptCBO_(cargo) {
  return 'Você é um especialista na Classificação Brasileira de Ocupações (CBO) do Ministério do Trabalho e Emprego do Brasil.\n\n' +
    'O usuário digitou o cargo: "' + cargo + '"\n\n' +
    'Identifique o código CBO mais adequado para este cargo e retorne SOMENTE um objeto JSON válido, sem nenhum texto antes ou depois, sem markdown, sem backticks. Formato exato:\n\n' +
    '{\n  "codigo": "XXXXXX",\n  "titulo": "Título oficial conforme CBO",\n  "descricao": "Descrição completa da ocupação conforme o MTE (2-4 frases)",\n' +
    '  "atribuicoes": ["Atribuição típica 1","Atribuição típica 2","Atribuição típica 3","Atribuição típica 4","Atribuição típica 5","Atribuição típica 6"],\n  "encontrado": true\n}\n\n' +
    'Se o cargo não existir na CBO ou for muito genérico para mapear, retorne: {"encontrado": false}\n\nResponda APENAS com o JSON, sem explicações.';
}

function construirPromptPDI_(f) {
  const compTexto = (f.competencias||[]).map(function(c){
    return '  - ' + c.nome + ': ' + c.val + '/5 (' + (NIVEL_LABELS_[c.val-1]||'não avaliado') + ')';
  }).join('\n');
  const interesseLabel = {'sim-migrar':'Sim, tem interesse em migrar de área','sim-manter':'Sim, tem interesse em outra área mas prefere permanecer','nao':'Não, quer crescer na área atual'};
  const disponLabel = {'horario-trabalho':'Durante o horário de trabalho','fora-horario':'Fora do horário de trabalho','ambos':'Ambos os horários'};

  return 'Você é um especialista em Gestão de Pessoas e Desenvolvimento Humano. Com base nas informações abaixo, elabore um PDI (Plano de Desenvolvimento Individual) completo, estruturado e personalizado para este colaborador.\n\n' +
    '## DADOS DO COLABORADOR\n' +
    '- Nome: ' + f.nome + '\n' +
    '- Cargo: ' + f.cargo + (f.cbo ? ' (CBO ' + f.cbo + ' — ' + f.cboTitulo + ')' : '') + '\n' +
    '- Departamento: ' + (f.depto||'—') + '\n' +
    '- Nível auto-avaliado: ' + (f.nivel||'não informado') + '\n' +
    '- Tempo de empresa: ' + (f.experiencia||'não informado') + '\n' +
    '- Formação: ' + (f.formacao||'não informado') + '\n' +
    '- Idiomas: ' + (f.idiomas||'não informado') + '\n' +
    '- Autonomia: ' + (f.autonomia||'não informado') + '\n\n' +
    '## ATRIBUIÇÕES\n' +
    '- O que COMPETE ao cargo: ' + (f.atribuicoesCompete||'não informado') + '\n' +
    '- O que faz ALÉM do cargo: ' + (f.atribuicoesAlem||'não informado') + '\n' +
    '- O que ainda NÃO executa: ' + (f.atribuicoesNaoFaz||'não informado') + '\n\n' +
    '## AUTO-AVALIAÇÃO DE COMPETÊNCIAS (escala 1-5)\n' + compTexto + '\n\n' +
    '## AUTODESENVOLVIMENTO\n' +
    '- Pontos fortes: ' + (f.pontosFortes||'não informado') + '\n' +
    '- Áreas para desenvolvimento: ' + (f.areasDesenvolvimento||'não informado') + '\n' +
    '- Expectativa de carreira (2 anos): ' + (f.expectativaCarreira||'não informado') + '\n' +
    '- Interesse em outras áreas: ' + (interesseLabel[f.interesse]||'não informado') + '\n' +
    '- Disponibilidade para capacitação: ' + (disponLabel[f.disponibilidade]||'não informado') + '\n\n---\n\n' +
    'Elabore o PDI seguindo EXATAMENTE esta estrutura:\n\n' +
    '**1. DIAGNÓSTICO DO COLABORADOR**\nResumo analítico do perfil atual, destacando pontos fortes, lacunas identificadas e potencial observado.\n\n' +
    '**2. OBJETIVOS DE DESENVOLVIMENTO (6 a 12 meses)**\nListe 3 a 5 objetivos específicos, mensuráveis e alinhados às lacunas e expectativas do colaborador.\n\n' +
    '**3. PLANO DE AÇÃO**\nPara cada objetivo, defina:\n- Ação concreta a ser realizada\n- Recurso / modalidade (curso, mentoria, job rotation, leitura, projeto prático, etc.)\n- Prazo sugerido\n- Indicador de conclusão\n\n' +
    '**4. COMPETÊNCIAS PRIORITÁRIAS**\nIndique as 3 competências com maior necessidade de desenvolvimento, com justificativa baseada nas notas e nos dados da ficha.\n\n' +
    '**5. RECOMENDAÇÕES PARA O RH**\nOrientações específicas para o gestor e o RH acompanharem o desenvolvimento deste colaborador, incluindo sugestões de enquadramento de nível se aplicável.\n\n' +
    'Seja objetivo, prático e personalizado. Use linguagem profissional mas acessível.';
}

function doGet(e) {
  if (e.parameter.action === 'listarFichas') {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const sheet = ss.getSheetByName(SHEET_NAME);
    const rows = sheet.getDataRange().getValues();
    const fichas = [];
    for (let i = 1; i < rows.length; i++) {
      if (!rows[i][0]) continue;
      fichas.push(rowToFicha_(rows[i]));
    }
    return ContentService
      .createTextOutput(JSON.stringify(fichas))
      .setMimeType(ContentService.MimeType.JSON);
  }

  // Busca CBO por IA — usado na Etapa 1 do formulário do colaborador
  if (e.parameter.action === 'buscarCBO') {
    try {
      const cargo = e.parameter.cargo || '';
      const raw = chamarClaude(construirPromptCBO_(cargo), 800);
      const clean = raw.replace(/^```json\s*/,'').replace(/^```\s*/,'').replace(/\s*```$/,'').trim();
      return ContentService.createTextOutput(clean).setMimeType(ContentService.MimeType.JSON);
    } catch (err) {
      return ContentService
        .createTextOutput(JSON.stringify({ encontrado:false, erro: err.message }))
        .setMimeType(ContentService.MimeType.JSON);
    }
  }

  // Geração de PDI por IA — usado no painel do RH
  if (e.parameter.action === 'gerarPDI') {
    try {
      const ss = SpreadsheetApp.getActiveSpreadsheet();
      const sheet = ss.getSheetByName(SHEET_NAME);
      const rows = sheet.getDataRange().getValues();
      let ficha = null;
      for (let i = 1; i < rows.length; i++) {
        if (String(rows[i][0]) === String(e.parameter.id)) { ficha = rowToFicha_(rows[i]); break; }
      }
      if (!ficha) throw new Error('Ficha não encontrada');
      const texto = chamarClaude(construirPromptPDI_(ficha), 2000);
      return ContentService
        .createTextOutput(JSON.stringify({ ok:true, texto:texto }))
        .setMimeType(ContentService.MimeType.JSON);
    } catch (err) {
      return ContentService
        .createTextOutput(JSON.stringify({ ok:false, error: err.message }))
        .setMimeType(ContentService.MimeType.JSON);
    }
  }

  return ContentService
    .createTextOutput(JSON.stringify({ ok:false, error:'Ação inválida' }))
    .setMimeType(ContentService.MimeType.JSON);
}

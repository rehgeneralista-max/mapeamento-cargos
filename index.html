⚙️ Configurar Google Sheets

📋 O que vai acontecer: qualquer colaborador que preencher a ficha em qualquer computador terá os dados salvos automaticamente na sua planilha Google Sheets. Você acessa o módulo RH de qualquer lugar e vê tudo.

PASSO 1 — Criar a planilha

  1. Acesse sheets.google.com e crie uma nova planilha em branco
  2. Renomeie a aba para Fichas
  3. Copie o ID da planilha da URL (parte entre /d/ e /edit)

PASSO 2 — Criar o Apps Script

  1. Na planilha, clique em Extensões → Apps Script
  2. Apague todo o código existente
  3. Cole o código abaixo e salve (Ctrl+S)
const SHEET_NAME = 'Fichas';

function doPost(e) {
  try {
    const data = JSON.parse(e.postData.contents);
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const sheet = ss.getSheetByName(SHEET_NAME);

    if (data.action === 'salvarFicha') {
      const f = data.ficha;
      const comps = (f.competencias||[]).map(c=>c.nome+':'+c.val).join('|');
      sheet.appendRow([
        f.id, f.nome, f.cargo, f.cbo, f.cboTitulo, f.depto,
        f.dataAdmissao, f.entrevistador, f.dataEntrevista,
        f.nivel, f.autonomia, f.formacao, f.experiencia, f.idiomas,
        f.atribuicoesCompete, f.atribuicoesAlem, f.atribuicoesNaoFaz,
        f.atividades, f.frequencia, f.cargaHoraria,
        f.responsabilidades, comps,
        f.pontosFortes, f.areasDesenvolvimento, f.expectativaCarreira,
        f.interesse, f.disponibilidade,
        f.data, f.status, '', '', ''
      ]);
    }

    if (data.action === 'atualizarFicha') {
      const sheet = ss.getSheetByName(SHEET_NAME);
      const rows = sheet.getDataRange().getValues();
      for (let i = 1; i < rows.length; i++) {
        if (String(rows[i][0]) === String(data.id)) {
          const campos = data.campos;
          const col = {
            status: 29, pdi: 30, prazoCtrl: 31, obsRH: 32
          };
          for (const [k, v] of Object.entries(campos)) {
            if (col[k]) sheet.getRange(i+1, col[k]).setValue(v);
          }
          break;
        }
      }
    }

    return ContentService
      .createTextOutput(JSON.stringify({ok:true}))
      .setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService
      .createTextOutput(JSON.stringify({ok:false,error:err.message}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function doGet(e) {
  if (e.parameter.action === 'listarFichas') {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const sheet = ss.getSheetByName(SHEET_NAME);
    const rows = sheet.getDataRange().getValues();
    const fichas = [];
    for (let i = 1; i < rows.length; i++) {
      const r = rows[i];
      if (!r[0]) continue;
      const comps = String(r[21]||'').split('|').filter(Boolean).map(c=>{
        const [nome,val] = c.split(':');
        return {nome, val:parseInt(val)||0};
      });
      fichas.push({
        id:String(r[0]), nome:r[1], cargo:r[2], cbo:r[3], cboTitulo:r[4],
        depto:r[5], dataAdmissao:r[6], entrevistador:r[7], dataEntrevista:r[8],
        nivel:r[9], autonomia:r[10], formacao:r[11], experiencia:r[12], idiomas:r[13],
        atribuicoesCompete:r[14], atribuicoesAlem:r[15], atribuicoesNaoFaz:r[16],
        atividades:r[17], frequencia:r[18], cargaHoraria:r[19],
        responsabilidades:r[20], competencias:comps,
        pontosFortes:r[22], areasDesenvolvimento:r[23], expectativaCarreira:r[24],
        interesse:r[25], disponibilidade:r[26],
        data:r[27], status:r[28]||'novo', pdi:r[29]||'',
        prazoCtrl:r[30]||'', obsRH:r[31]||''
      });
    }
    return ContentService
      .createTextOutput(JSON.stringify(fichas))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
